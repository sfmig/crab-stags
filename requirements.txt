caliscope==0.11.5

# caliscope declares only `opencv-python>=4.8.0.74` (no upper bound), so a fresh
# resolve picks up OpenCV 5.x. OpenCV 5 moved the fisheye calibration flags out of
# the cv2.fisheye submodule to the top level, so caliscope's fisheye intrinsic
# calibration dies with:
#   AttributeError: module 'cv2.fisheye' has no attribute 'CALIB_RECOMPUTE_EXTRINSIC'
# Cap at <5 until caliscope supports OpenCV 5.
opencv-python>=4.8.0.74,<5

numpy==2.5.1
stag-python==1.1.1
